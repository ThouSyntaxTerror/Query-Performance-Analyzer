2. WHO INSERTS ROWS (answer + proof)
------------------------------------
Say: "Admins insert companies, jobs, students and placements. Students insert their own applications through a
      stored procedure. Triggers insert the audit log. Every row stores created_by, the user who inserted it."

  SELECT 'jobs' AS tbl, u.username, COUNT(*) AS rows_inserted FROM jobs j JOIN users u ON u.user_id = j.created_by GROUP BY u.username;
  SELECT 'applications' AS tbl, u.role, COUNT(*) AS rows_inserted FROM applications a JOIN users u ON u.user_id = a.created_by GROUP BY u.role;
  SELECT audit_id, table_name, action, old_value, new_value, changed_by FROM audit_log ORDER BY audit_id DESC LIMIT 5;

6. JOIN, GROUP BY, HAVING
-------------------------
  SELECT j.title, c.name AS company, COUNT(*) AS applications FROM jobs j JOIN companies c ON c.company_id = j.company_id JOIN applications a ON a.job_id = j.job_id GROUP BY j.job_id, j.title, c.name HAVING COUNT(*) >= 80 ORDER BY applications DESC LIMIT 3;
  SELECT status, COUNT(*) AS n FROM applications GROUP BY status;


7. CASCADE (inside a transaction, nothing is lost)
--------------------------------------------------
  START TRANSACTION;
  SELECT COUNT(*) AS applications_before FROM applications WHERE student_id = 1;
  DELETE FROM students WHERE student_id = 1;
  SELECT COUNT(*) AS applications_after FROM applications WHERE student_id = 1;
  ROLLBACK;
  SELECT COUNT(*) AS applications_restored FROM applications WHERE student_id = 1;
